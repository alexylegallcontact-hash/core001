        // Gestion de la soumission de la demande (Option 2)

        function handleRequestSubmit(event) {

            event.preventDefault();

            const requestZone = document.getElementById('requestZone');

            const confirmZone = document.getElementById('confirmZone');


            // Animation de sortie

            requestZone.classList.add('opacity-0', 'scale-95');

            

            setTimeout(() => {

                requestZone.classList.add('hidden');

                confirmZone.classList.remove('hidden');

                

                // Animation d'entrée du message de succès

                setTimeout(() => {

                    confirmZone.classList.remove('opacity-0', 'translate-y-4');

                }, 50);

            }, 300);

        }


        // Réinitialiser la barre de recherche

        function resetInterface() {

            const requestZone = document.getElementById('requestZone');

            const confirmZone = document.getElementById('confirmZone');

            

            confirmZone.classList.add('opacity-0', 'translate-y-4');

            

            setTimeout(() => {

                confirmZone.classList.add('hidden');

                requestZone.classList.remove('hidden');

                document.getElementById('userRequest').value = '';

                

                setTimeout(() => {

                    requestZone.classList.remove('opacity-0', 'scale-95');

                }, 50);

            }, 300);

        }


        // Ouvrir / Fermer la fenêtre de connexion/inscription

        function toggleAuthModal(mode = null) {

            const modal = document.getElementById('authModal');

            const title = document.getElementById('modalTitle');

            const subtitle = document.getElementById('modalSubtitle');

            const btn = document.getElementById('modalSubmitBtn');


            if (!mode) {

                modal.classList.add('opacity-0');

                setTimeout(() => modal.classList.add('hidden'), 200);

                return;

            }


            if (mode === 'login') {

                title.innerText = 'Se connecter';

                subtitle.innerText = 'Accédez à votre espace unique.';

                btn.innerText = 'Se connecter';

            } else {

                title.innerText = 'Créer un compte';

                subtitle.innerText = 'Rejoignez-nous en quelques secondes.';

                btn.innerText = "S'inscrire";

            }


            modal.classList.remove('hidden');

            setTimeout(() => modal.classList.remove('opacity-0'), 10);

        }


        // Simulation de soumission Auth

        function handleAuthSubmit(event) {

            event.preventDefault();

            alert('Authentification réussie (Simulation)');

            toggleAuthModal();

        }
